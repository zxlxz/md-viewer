from dataclasses import dataclass


@dataclass
class Viewer:
    name: str = "md-viewer"
    themes: int = 4

    def render(self, source: str) -> str:
        """Render markdown into a reading view."""
        if not source:
            raise ValueError("empty document")
        return f"<article>{source}</article>"
